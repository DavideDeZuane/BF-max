-------- PARAMETERS

decoder: BF-Max
itMax: 201
r: 10499
discarded bits: 7
samples: 10^6

-------- EXPERIMENT 1

f_seed_keygen = $
f_pk, f_sk = keygen(f_seed_keygen)

for t in tests:

  v_seed_encaps = $
  v_seed_keygen = $

  // fixed keypair
  ct, ss = encaps(v_seed_encaps, f_pk)
  decaps(ct, f_sk) // measure

  // variable keypair
  v_pk, v_sk = keygen(v_seed_keygen)
  ct, ss = encaps(v_seed_encaps, v_pk)
  decaps(ct, v_sk) // measure

-------- EXPERIMENT 1 RESULTS

I(F;V) = 0.35184 bits
H(F) = 8.662 bits, H(V) = 8.670 bits
MI / min(H) = 4.0620%
